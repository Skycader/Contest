#!/bin/bash

# Находим все файлы по маске
files=(ex[0-9]{1,}.js)

# Проверяем, есть ли файлы, соответствующие маске
if [ ${#files[@]} -eq 0 ]; then
    echo "Нет файлов, соответствующих маске ex*.js"
    exit 1
fi

# Генерируем случайный индекс для выбора файла
random_index=$(($RANDOM % ${#files[@]}))

# Получаем случайный файл
random_file=${files[$random_index]}

# Открываем файл (например, в текстовом редакторе nano)
nvim "$random_file"

# Очищаем файл, если он уже существует
> all.txt

# Проходим по всем файлам в текущей директории
for file in *; do
    # Пропускаем, если это не обычный файл или это наш же all.txt
    [[ -f "$file" ]] || continue
    [[ "$file" == "all.txt" ]] && continue

    # Выводим разделитель + имя файла
    {
        echo "═══════════════════════════════════════════════════════════════"
        echo "Файл: $file"
        echo "═══════════════════════════════════════════════════════════════"
        echo ""

        # Выводим содержимое файла
        cat "$file"

        # Пустая строка после содержимого для читаемости
        echo ""
        echo ""

    } >> all.txt
done
